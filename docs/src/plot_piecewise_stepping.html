src.plot_piecewise_stepping

plot_piecewise_stepping.py

Manually-prescribed TWO-PHASE step size (fine during luminous flight, coarse during dark flight), as a hand-crafted alternative to a fully adaptive solver -- tests whether allocating resolution by trajectory phase is as effective as letting RK45 choose its own steps.

Methods compared: RK2, RK4 (literal fixed steps per phase), RK45 (adaptive, with phase-specific max_step caps). DOP853 (tight tolerance) is the numerical reference solution, not the physical truth.

Fine/coarse timestep pairs tested (coarse = 10x fine): (0.5, 5.0), (0.25, 2.5), (0.10, 1.0), (0.05, 0.5), (0.025, 0.25) s.

Reports, per method and step-size pair: impact-location error, trajectory error vs. the reference, stability/divergence at coarse steps, and computation time -- i.e. whether phase-aware stepping beats uniform stepping for the same cost.

 118def run_piecewise_fixed(method, dt_fine, dt_coarse):
 119    """
 120    RK2/RK4 piecewise fixed-step integration.
 121
 122    Phase 1:
 123        0 -> PHASE_SPLIT using dt_fine
 124
 125    Phase 2:
 126        PHASE_SPLIT -> FULL_END using dt_coarse
 127    """
 128
 129    f = lambda t, y: meteor_rhs(t, y, params=params)
 130
 131    y0 = initial_state()
 132
 133    phase1 = integrate_fixed_step(
 134        f,
 135        y0,
 136        0.0,
 137        PHASE_SPLIT,
 138        dt_fine,
 139        method=method,
 140    )
 141
 142    y_mid = phase1["y"][:, -1]
 143
 144    phase2 = integrate_fixed_step(
 145        f,
 146        y_mid,
 147        PHASE_SPLIT,
 148        FULL_END,
 149        dt_coarse,
 150        method=method,
 151    )
 152
 153    t_all = np.concatenate([
 154        phase1["t"],
 155        phase2["t"][1:],
 156    ])
 157
 158    y_all = np.concatenate([
 159        phase1["y"],
 160        phase2["y"][:, 1:],
 161    ], axis=1)
 162
 163    event_y = phase2["event_y"]
 164
 165    return (
 166        t_all,
 167        y_all,
 168        event_y,
 169        phase1["t"],
 170        phase1["y"],
 171    )

 180    RK45 piecewise integration.

 195        method="RK45",
 196        rtol=RK45_RTOL,
 197        atol=RK45_ATOL,
 198        max_step=max_step_fine,
 199    )
 200
 201    y_mid = sol1.y[:, -1]
 202
 203    ground_impact_event.terminal = True
 204    ground_impact_event.direction = -1
 205
 206    sol2 = solve_ivp(
 207        fun=lambda t, y: meteor_rhs(t, y, params=params),
 208        t_span=(PHASE_SPLIT, FULL_END),
 209        y0=y_mid,
 210        method="RK45",
 211        rtol=RK45_RTOL,
 212        atol=RK45_ATOL,
 213        max_step=max_step_coarse,
 214        events=ground_impact_event,
 215    )

 218        sol1.t,
 219        sol2.t[1:],
 220    ])

 223        sol1.y,

 262def numerical_jacobian(f, t, y, eps_rel=1e-6):
 263    """
 264    Finite-difference Jacobian df/dy of the RHS at (t, y), shape (7,7).
 265    Forward difference, step size scaled to each state's own magnitude
 266    to keep the perturbation well-conditioned across variables of very
 267    different scale (h in metres vs. gamma in radians, etc.).
 268    """
 269    n = len(y)
 270    J = np.zeros((n, n))
 271    f0 = f(t, y)
 272    for j in range(n):
 273        dy = np.zeros(n)
 274        step = eps_rel * max(abs(y[j]), 1.0)
 275        dy[j] = step
 276        f1 = f(t, y + dy)
 277        J[:, j] = (f1 - f0) / step
 278    return J

 281def local_timescale(f, t, y):
 282    """
 283    Rigorous local stability timescale: tau = 1 / max(|Re(eigenvalue)|)
 284    of the Jacobian df/dy at this state. This is the general definition
 285    -- no assumption about which state variable or physical effect
 286    dominates, unlike the hand-derived estimates used elsewhere in this
 287    project (scale-height/velocity for the luminous phase, drag
 288    sensitivity for dark flight). An explicit fixed-step method needs
 289    dt well below this value at every point along the trajectory to
 290    remain numerically stable.
 291    """
 292    J = numerical_jacobian(f, t, y)
 293    eigvals = np.linalg.eigvals(J)
 294    max_mag = np.max(np.abs(eigvals.real))
 295    return 1.0 / max_mag if max_mag > 0 else np.inf

 343def find_stiffest_time(data, t_lo, t_hi, n=400):
 344    """
 345    Scan [t_lo, t_hi] along the reference trajectory and return the time
 346    at which local_timescale() is SMALLEST (i.e. the Jacobian's dominant
 347    eigenvalue magnitude is LARGEST) -- the single worst-case instant for
 348    explicit-method stability in that window.
 349
 350    tau(t) is not monotonic (see figure_stability_timescale()'s plot),
 351    so this is not the same as the phase boundary or either endpoint --
 352    it has to be found by scanning.
 353    """
 354    reference = data["reference"]
 355    f = lambda t, y: meteor_rhs(t, y, params=params)
 356    t_vals = np.linspace(t_lo, t_hi, n)
 357    tau_vals = np.array([local_timescale(f, t, reference.sol(t)) for t in t_vals])
 358    return t_vals[np.argmin(tau_vals)]

 361def rk_stability_function(A, b, z):
 362    """
 363    Exact stability function R(z) of an explicit Runge-Kutta method with
 364    Butcher tableau (A, b), on the scalar linear test equation
 365    y' = lambda*y (z = h*lambda). Derived by forward substitution
 366    through the stages (A is strictly lower triangular for an explicit
 367    method, so no matrix inversion is needed): stage i has value
 368    Y_i = y_n*phi_i(z), with phi_i = 1 + z*sum_{j<i} A[i,j]*phi_j
 369    (phi_1 = 1 since c_1 = 0), and R(z) = 1 + z*sum_i b_i*phi_i(z).
 370    Standard result -- see Hairer & Wanner, "Solving Ordinary
 371    Differential Equations I", Ch. IV.2.
 372    """
 373    z = np.asarray(z, dtype=float)
 374    s = len(b)
 375    phi = [np.ones_like(z)]
 376    for i in range(1, s):
 377        acc = np.ones_like(z)
 378        for j in range(i):
 379            if A[i, j] != 0.0:
 380                acc = acc + z * A[i, j] * phi[j]
 381        phi.append(acc)
 382    R = np.ones_like(z)
 383    for i in range(s):
 384        if b[i] != 0.0:
 385            R = R + z * b[i] * phi[i]
 386    return R

 389def R_euler(z):
 390    """Stability function of explicit (forward) Euler."""
 391    return 1.0 + z

 394def R_rk2(z):
 395    """Stability function of Heun's method (explicit RK2)."""
 396    return 1.0 + z + z**2 / 2.0

 399def R_rk4(z):
 400    """Stability function of the classical 4th-order Runge-Kutta method."""
 401    return 1.0 + z + z**2 / 2.0 + z**3 / 6.0 + z**4 / 24.0

 404def R_implicit_euler(z):
 405    """
 406    Stability function of backward (implicit) Euler: R(z) = 1/(1-z).
 407    A genuine rational function (Q(z)=1-z != 1), unlike the explicit
 408    methods above -- which is exactly why it stays bounded as z -> -inf
 409    (|R(z)| -> 0) instead of blowing up like a polynomial does.
 410    """
 411    return 1.0 / (1.0 - z)
 412
 413
 414def R_implicit_heun(z):
 415    """
 416    Stability function of the implicit (trapezoidal) Heun method:
 417    R(z) = (1 + z/2)/(1 - z/2).
 418
 419    A-stable like implicit Euler (|R(z)|<=1 for all Re(z)<=0, so never
 420    unstable), but NOT L-stable: |R(z)| -> 1 as z -> -infinity, instead
 421    of -> 0. Very stiff modes are therefore preserved (as bounded
 422    oscillation) rather than damped out -- the reason it survives large
 423    steps worse than implicit Euler in practice despite being
 424    unconditionally stable in the same linear sense.
 425    """
 426    return (1.0 + z / 2.0) / (1.0 - z / 2.0)

 476def figure_amplification_factor(data, t_sample=None, n_h=400):
 477    """
 478    Amplification factor |A(h)| vs. step size h, for RK2, RK4 and RK45,
 479    all evaluated at ONE representative local eigenvalue
 480    kappa = 1/tau(t_sample) (same numerical_jacobian()/local_timescale()
 481    machinery used for the exact stability-boundary analysis elsewhere
 482    in this file).
 483
 484    This is the vector-system extension of the notes' scalar
 485    perturbation-based stability derivation: the local Jacobian's
 486    dominant eigenvalue plays the role of the notes' single lambda
 487    (lambda = -kappa, a decaying real mode), and each method's
 488    amplification factor is evaluated at z = h*lambda. Each method's
 489    |R(z)| exceeds 1 beyond its own stability boundary, visible
 490    directly as the curve crossing the dashed threshold line.
 491
 492    Default t_sample: the STIFFEST instant in the luminous phase (the
 493    minimum of tau(t) over [0, PHASE_SPLIT], found by find_stiffest_time()
 494    -- NOT the phase-split time itself). tau(t) dips to its smallest
 495    value mid-luminous-phase (around peak dynamic pressure/heating) and
 496    is actually larger again by the time dark flight starts, so this is
 497    the point that genuinely sets the binding stability constraint the
 498    fine step size has to respect -- an arbitrary point like PHASE_SPLIT
 499    understates how tight that constraint really is.
 500    """
 501    reference = data["reference"]
 502    if t_sample is None:
 503        t_sample = find_stiffest_time(data, 0.0, PHASE_SPLIT)
 504    f = lambda t, y: meteor_rhs(t, y, params=params)
 505    y_sample = reference.sol(t_sample)
 506    kappa = 1.0 / local_timescale(f, t_sample, y_sample)
 507
 508    h_vals = np.logspace(-2, 3, n_h)
 509    z_vals = -kappa * h_vals  # lambda = -kappa (decaying real mode)
 510
 511    curves = {
 512        "RK2": np.abs(R_rk2(z_vals)),
 513        "RK4": np.abs(R_rk4(z_vals)),
 514        "RK45": np.abs(R_rk45(z_vals)),
 515        "Implicit Euler": np.abs(R_implicit_euler(z_vals)),
 516        "Implicit Heun": np.abs(R_implicit_heun(z_vals)),
 517    }
 518
 519    fig, ax = plt.subplots(figsize=(9, 6.5))
 520    for label, A_vals in curves.items():
 521        ax.loglog(h_vals, A_vals, "-", color=METHOD_COLORS.get(label, "gray"),
 522                   linewidth=2, label=label)
 523
 524    ax.axhline(1.0, color="black", linestyle="--", linewidth=1,
 525                label="stability threshold |A|=1")
 526    ax.set_xlabel("Step size h (s)")
 527    ax.set_ylabel("Amplification factor |A(h)|")
 528    ax.set_title(
 529        f"Amplification factor vs. step size at t={t_sample:g}s (stiffest luminous-phase instant)\n"
 530        f"kappa={kappa:.4g} /s, tau={1/kappa:.4g}s"
 531    )
 532    ax.grid(True, which="both", linestyle="--", alpha=0.4)
 533    ax.legend()
 534    plt.tight_layout()
 535    plt.show()

 542def path_is_sane(
 543    y_array,
 544    lat_idx=3,
 545    lon_idx=4,
 546    bound_deg=90,
 547):
 548    """
 549    Numerical validity check on an integrated path.
 550
 551    Returns False if:
 552        - NaNs/infinities occur
 553        - latitude exceeds physical bounds
 554        - longitude exceeds physical bounds
 555        - height becomes absurdly large
 556    """
 557
 558    if not np.all(np.isfinite(y_array)):
 559        return False
 560
 561    lat_deg = np.degrees(y_array[lat_idx, :])
 562    lon_deg = np.degrees(y_array[lon_idx, :])
 563
 564    if not np.all(np.abs(lat_deg) <= bound_deg):
 565        return False
 566
 567    if not np.all(np.abs(lon_deg) <= 180):
 568        return False
 569
 570    height_km = np.abs(y_array[0, :]) / 1000.0
 571
 572    if np.any(height_km > HEIGHT_LIMIT_KM):
 573        return False
 574
 575    return True

 582def interpolate_reference(reference, t):
 583    """
 584    Evaluate the dense DOP853 reference solution at arbitrary times.
 585    """
 586
 587    t_clipped = np.clip(
 588        t,
 589        reference.t[0],
 590        reference.t[-1],
 591    )
 592
 593    return reference.sol(t_clipped)

 600def calculate_trajectory_error(t, y, reference):
 601    """
 602    Calculate trajectory/state error relative to the reference solution.

 645# COMPUTE ALL RESULTS
 646# ============================================================================
 647
 648def compute_piecewise_results():
 649    """
 650    Run the reference solution and every method/timestep combination.
 651
 652    Results are retained even when a trajectory becomes unstable so that
 653    the divergence itself can be plotted and analysed. Wall-clock runtime
 654    is recorded per run for the computation-time-vs-accuracy comparison.
 655    """
 656
 657    print("\n" + "=" * 80)
 658    print("REFERENCE SOLUTION")
 659    print("=" * 80)
 660
 661    reference = run_reference()
 662
 663    if len(reference.t_events[0]) > 0:
 664        reference_event_y = reference.y_events[0][0]
 665        reference_impact_time = reference.t_events[0][0]
 666
 667        lat_ref = np.degrees(reference_event_y[3])
 668        lon_ref = np.degrees(reference_event_y[4])
 669
 670        print(f"Reference impact time : {reference_impact_time:.8f} s")
 671        print(f"Reference latitude    : {lat_ref:.8f} deg")
 672        print(f"Reference longitude   : {lon_ref:.8f} deg")
 673
 674    else:
 675        reference_event_y = None
 676        reference_impact_time = None
 677        print("WARNING: Reference solution did not reach ground.")
 678
 679    t_reference_plot = np.linspace(0, reference.t[-1], 1000)
 680    y_reference_plot = reference.sol(t_reference_plot)
 681    height_reference_plot = y_reference_plot[0] / 1000.0
 682
 683    # Implicit Euler gets a few extra, much larger dt_coarse values on
 684    # top of the shared PAIRS: being unconditionally stable (in the
 685    # local-linear sense), it can take coarse steps RK2/RK4/RK45 simply
 686    # cannot survive -- these extra pairs are what actually demonstrates
 687    # that advantage in the computation-time vs. impact-error plot,
 688    # rather than only asserting it. 20.0 s was found empirically to be
 689    # about where the step is still both Newton-stable and physically
 690    # valid for this trajectory; 50.0 s+ already produces nonphysical
 691    # (even negative) velocities despite "converging" -- unconditional
 692    # stability is not the same as accuracy.
 693    IMPLICIT_EXTRA_PAIRS = [(0.5, 10.0), (0.5, 15.0), (0.5, 20.0)]
 694
 695    methods = [
 696        ("RK2", "fixed", "rk2", PAIRS),
 697        ("RK4", "fixed", "rk4", PAIRS),
 698        ("RK45", "rk45", None, PAIRS),
 699        ("Implicit Euler", "fixed", "implicit_euler", PAIRS + IMPLICIT_EXTRA_PAIRS),
 700        ("Implicit Heun", "fixed", "implicit_heun", PAIRS + IMPLICIT_EXTRA_PAIRS),
 701    ]
 702
 703    results_by_method = []
 704    all_lons = []
 705    all_lats = []
 706
 707    if reference_event_y is not None:
 708        all_lons.append(np.degrees(reference_event_y[4]))
 709        all_lats.append(np.degrees(reference_event_y[3]))
 710
 711    all_lons.append(RECOVERY_LON_DEG)
 712    all_lats.append(RECOVERY_LAT_DEG)
 713
 714    for label, kind, fkey, pairs in methods:
 715
 716        entries = []
 717        phase1_entries = []
 718        raw_full = []
 719        failed = []
 720        colors = plt.cm.plasma(np.linspace(0, 0.8, len(pairs)))
 721
 722        for (dt_fine, dt_coarse), color in zip(pairs, colors):
 723
 724            # ------------------------------------------------------
 725            # Run (timed)
 726            # ------------------------------------------------------
 727
 728            t0 = time.perf_counter()
 729
 730            # The largest coarse steps (particularly for Implicit Euler/
 731            # Heun's extra pairs) are expected to produce an unstable or
 732            # non-impacting run, which triggers numpy overflow warnings
 733            # inside dynamics.py. That outcome is already captured by
 734            # this run's own "status" (UNSTABLE/NO IMPACT) below, so the
 735            # warnings themselves are suppressed.
 736            with warnings.catch_warnings():
 737                warnings.simplefilter("ignore")
 738                if kind == "fixed":
 739                    (t_all, y_all, event_y, t_p1, y_p1) = run_piecewise_fixed(
 740                        fkey, dt_fine, dt_coarse,
 741                    )
 742                else:
 743                    (t_all, y_all, event_y, t_p1, y_p1) = run_piecewise_rk45(
 744                        dt_fine, dt_coarse,
 745                    )
 746
 747            runtime_s = time.perf_counter() - t0
 748
 749            phase1_entries.append((dt_fine, dt_coarse, t_p1, y_p1, color))
 750            raw_full.append((dt_fine, dt_coarse, t_all, y_all, color))
 751
 752            sane = path_is_sane(y_all)
 753            impact_error = calculate_impact_error(event_y, reference_event_y)
 754
 755            if np.all(np.isfinite(y_all)):
 756                errors = calculate_trajectory_error(t_all, y_all, reference)
 757                height_error = errors[0, :]
 758                max_height_error = np.nanmax(height_error)
 759            else:
 760                errors = None
 761                max_height_error = np.inf
 762
 763            n_steps = len(t_all) - 1
 764
 765            if event_y is None:
 766                status = "NO IMPACT"
 767                failed.append((dt_fine, dt_coarse))
 768            elif not sane:
 769                status = "UNSTABLE"
 770                failed.append((dt_fine, dt_coarse))
 771            else:
 772                status = "OK"
 773
 774            if event_y is not None and sane:
 775                lat_i = event_y[3]
 776                lon_i = event_y[4]
 777                all_lons.append(np.degrees(lon_i))
 778                all_lats.append(np.degrees(lat_i))
 779
 780            entries.append({
 781                "dt_fine": dt_fine,
 782                "dt_coarse": dt_coarse,
 783                "t": t_all,
 784                "y": y_all,
 785                "event_y": event_y,
 786                "color": color,
 787                "sane": sane,
 788                "status": status,
 789                "impact_error": impact_error,
 790                "trajectory_errors": errors,
 791                "max_height_error": max_height_error,
 792                "n_steps": n_steps,
 793                "runtime_s": runtime_s,
 794            })
 795
 796        results_by_method.append({
 797            "label": label,
 798            "entries": entries,
 799            "phase1_entries": phase1_entries,
 800            "raw_full": raw_full,
 801            "failed": failed,
 802        })
 803
 804        if failed:
 805            print(f"\n{label}: {len(failed)} pair(s) failed or became unstable.")
 806
 807    return {
 808        "reference": reference,
 809        "reference_event_y": reference_event_y,
 810        "reference_impact_time": reference_impact_time,
 811        "t_reference_plot": t_reference_plot,
 812        "height_reference_plot": height_reference_plot,
 813        "results_by_method": results_by_method,
 814        "all_lons": all_lons,
 815        "all_lats": all_lats,
 816    }

def numerical_jacobian(f, t, y, eps_rel=1e-06): 
263def numerical_jacobian(f, t, y, eps_rel=1e-6):
264    """
265    Finite-difference Jacobian df/dy of the RHS at (t, y), shape (7,7).
266    Forward difference, step size scaled to each state's own magnitude
267    to keep the perturbation well-conditioned across variables of very
268    different scale (h in metres vs. gamma in radians, etc.).
269    """
270    n = len(y)
271    J = np.zeros((n, n))
272    f0 = f(t, y)
273    for j in range(n):
274        dy = np.zeros(n)
275        step = eps_rel * max(abs(y[j]), 1.0)
276        dy[j] = step
277        f1 = f(t, y + dy)
278        J[:, j] = (f1 - f0) / step
279    return J

283    """
284    Rigorous local stability timescale: tau = 1 / max(|Re(eigenvalue)|)
285    of the Jacobian df/dy at this state. This is the general definition
286    -- no assumption about which state variable or physical effect
287    dominates, unlike the hand-derived estimates used elsewhere in this
288    project (scale-height/velocity for the luminous phase, drag

290    dt well below this value at every point along the trajectory to
291    remain numerically stable.
292    """

363    """
364    Exact stability function R(z) of an explicit Runge-Kutta method with
365    Butcher tableau (A, b), on the scalar linear test equation
366    y' = lambda*y (z = h*lambda). Derived by forward substitution
367    through the stages (A is strictly lower triangular for an explicit
368    method, so no matrix inversion is needed): stage i has value
369    Y_i = y_n*phi_i(z), with phi_i = 1 + z*sum_{j<i} A[i,j]*phi_j
370    (phi_1 = 1 since c_1 = 0), and R(z) = 1 + z*sum_i b_i*phi_i(z).
371    Standard result -- see Hairer & Wanner, "Solving Ordinary
372    Differential Equations I", Ch. IV.2.
373    """
374    z = np.asarray(z, dtype=float)
375    s = len(b)
376    phi = [np.ones_like(z)]
377    for i in range(1, s):
378        acc = np.ones_like(z)
379        for j in range(i):
380            if A[i, j] != 0.0:
381                acc = acc + z * A[i, j] * phi[j]
382        phi.append(acc)
383    R = np.ones_like(z)
384    for i in range(s):
385        if b[i] != 0.0:
386            R = R + z * b[i] * phi[i]
387    return R
